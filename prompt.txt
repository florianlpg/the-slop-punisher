## Key Implementation Patterns

### 1. Types Definition (`app/rules/types.ts`)
Types are defined in the app directory, not in components:
```typescript
export type RuleTableRow = {
  _id: Id<"rules">
  description: string
  fineAmountCents: number
  unit: RuleUnit
  // ... other fields
  totalMembers: number // merged from external source
}
```

### 2. Column Helper (`columns/helper.ts`)
```typescript
import { createColumnHelper } from "@tanstack/react-table"
import { type DataTableFeatures } from "@/components/Rules/RulesTable/RulesTableFeatures"
import { type RuleTableRow } from "@/app/rules/types"

export const columnHelper = createColumnHelper<DataTableFeatures, RuleTableRow>()
```

### 3. Column Index (`columns/index.ts`)
```typescript
import { columnHelper } from "./helper"
import { descriptionColumn } from "./description"
import { fineColumn } from "./fine"
// ... other imports

export const columns = columnHelper.columns([
  descriptionColumn,
  fineColumn,
  votesColumn,
  requiredApprovalsColumn,
  actionsColumn,
])
```

### 4. Individual Column Pattern (`columns/description.tsx`)
```typescript
"use client"
import { columnHelper } from "./helper"

export const descriptionColumn = columnHelper.accessor("description", {
  header: "Rule",
  cell: ({ getValue }) => (
    <div className="max-w-[420px] whitespace-pre-wrap">{getValue()}</div>

```

### 5. Complex Cell Component (`columns/actions.tsx`)
For interactive cells, create a separate component:
```typescript
function ActionsCell({ rule }: { rule: RuleTableRow }) {
  const archiveRule = useMutation(api.rules.archive)
  // ... implementation
}

export const actionsColumn = columnHelper.display({
  id: "actions",
  cell: ({ row }) => <ActionsCell rule={row.original} />,
})
```

### 6. Table Features (`RulesTableFeatures.tsx`)
```typescript
import {
  columnFilteringFeature,
  columnVisibilityFeature,
  createFilteredRowModel,
  createPaginatedRowModel,
  createSortedRowModel,
  filterFn_includesString,
  rowPaginationFeature,
  rowSortingFeature,
  sortFn_alphanumeric,
  sortFn_text,
  tableFeatures,
} from "@tanstack/react-table"

export const features = tableFeatures({
  columnFilteringFeature,
  columnVisibilityFeature,
  rowPaginationFeature,
  rowSortingFeature,
  filteredRowModel: createFilteredRowModel(),
  paginatedRowModel: createPaginatedRowModel(),
  sortedRowModel: createSortedRowModel(),
  filterFns: { includesString: filterFn_includesString },
  sortFns: { alphanumeric: sortFn_alphanumeric, text: sortFn_text },
})

export type DataTableFeatures = typeof features
```

### 7. Main Table Component (`RulesTable.tsx`)
```typescript
"use client"
import * as React from "react"
import { useTable, type SortingState } from "@tanstack/react-table"
import { useQuery } from "convex/react"
import { api } from "@/convex/_generated/api"
import { columns } from "./columns"
import { features } from "./RulesTableFeatures"
import { type RuleTableRow } from "@/app/rules/types"

export function RulesTable() {
  const rules = useQuery(api.rules.list)
  const totalMembers = useQuery(api.users.count)

  const data: RuleTableRow[] = rules?.map((rule) => ({
    ...rule,
    totalMembers: totalMembers ?? 0,
  })) ?? []

  const [sorting, setSorting] = React.useState<SortingState>([])

  const table = useTable({
    features,
    data,
    columns,
    onSortingChange: setSorting,
    state: { sorting },
  })

  // Render table with skeleton loading states
}
```

## For Approbations Feature

For the Approbations feature, follow this exact pattern:
1. Create `app/approbations/types.ts` with `ApprobationTableRow` type
2. Mirror the column structure with fields like: `accused`, `amount`, `description`, `createdAt`, `approvals`, `actions`
3. Use the same TanStack setup with `columnHelper`, `features`, and `useTable`
4. Implement skeleton loading states the same way
5. Use Convex mutations for actions (approve/reject)
6. Follow the naming convention: `ApprobationsTable.tsx`, `ApprobationsTableFeatures.tsx`

This will ensure consistency with your existing Rules implementation.
